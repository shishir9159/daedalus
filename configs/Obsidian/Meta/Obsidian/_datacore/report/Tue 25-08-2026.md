function splitNote(body) {

const WEEKLY_DIR = "Meta/Tracker/Weekly";
const WD_LONG = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];
const WD_ABBR = ["sun", "mon", "tue", "wed", "thu", "fri", "sat"];
const TASK_LINE = /^\s*[-*+]\s+\[.\]/;

function weekdayOf(file) {
  const m = String(file.basename).match(/\b(sun|mon|tue|wed|thu|fri|sat)[a-z]*\b/i);
  if (m) return WD_LONG[WD_ABBR.indexOf(m[1].toLowerCase())];
  return WD_LONG[new Date(file.stat ? file.stat.ctime : Date.now()).getDay()];
}

/* Headings of a note, keyed by normalised name -> line index.  Fenced blocks
   are skipped so a `#` inside this very code block cannot pose as a section. */
function headIndex(lines) {
  const map = new Map();
  let fence = false;
  lines.forEach((l, i) => {
    if (/^\s*(```|~~~)/.test(l)) { fence = !fence; return; }
    if (fence) return;
    const h = l.match(HEAD);
    if (!h) return;
    const k = norm(unlink(h[1]));
    if (k && !map.has(k)) map.set(k, i);
  });
  return map;
}

/* Returns { daily, lines } — the rewritten daily note and the raw source
   lines that were moved — or null when there is nothing to move. */
function pullPlan(srcText, dstText) {
  const src = srcText.split("\n");
  const dst = dstText.split("\n");
  const dstHead = headIndex(dst);

  const taken = [], drops = new Map();
  let target = null, fence = false;
  for (const raw of src) {
    if (/^\s*(```|~~~)/.test(raw)) { fence = !fence; continue; }
    if (fence) continue;
    const h = raw.match(HEAD);
    if (h) {
      const k = norm(unlink(h[1]));
      target = dstHead.has(k) ? dstHead.get(k) : null;
      continue;
    }
    if (target == null || !TASK_LINE.test(raw)) continue;
    if (!drops.has(target)) drops.set(target, []);
    drops.get(target).push(raw);
    taken.push(raw);
  }
  if (!taken.length) return null;

  /* insert bottom-up so the earlier heading indices stay valid */
  Array.from(drops.keys()).sort((a, b) => b - a).forEach(h => {
    let at = h + 1;
    while (at < dst.length && !dst[at].trim()) at++;
    dst.splice(at, 0, ...drops.get(h));
  });

  return { daily: dst.join("\n"), lines: taken };
}

const dropOnce = (text, lines) => {
  const out = text.split("\n");
  lines.forEach(l => { const i = out.indexOf(l); if (i >= 0) out.splice(i, 1); });
  return out.join("\n");
};

async function pullWeekly(app, file) {
  const dir = app.vault.getAbstractFileByPath(WEEKLY_DIR);
  if (!dir || !dir.children) return false;
  const want = norm(weekdayOf(file));
  const sources = dir.children.filter(c =>
    !c.children && /\.md$/i.test(c.name) && norm(c.basename).split(" ").includes(want));

  let moved = false;
  for (const src of sources) {
    const srcText = await app.vault.read(src);
    let taken = null;
    /* process() re-reads under the vault's own lock, so an editor with unsaved
       changes in either note is never clobbered. */
    await app.vault.process(file, data => {
      const plan = pullPlan(srcText, data);
      if (!plan) return data;
      taken = plan.lines;
      return plan.daily;
    });
    if (!taken) continue;
    await app.vault.process(src, data => dropOnce(data, taken));
    moved = true;
  }
  return moved;
}

/* ── Iconize glyphs ───────────────────────────────────────────────────────
   Iconize unpacks its downloaded packs asynchronously, so on a cold start
   the first render of :RaRoundShield: / :BoBxsMobileVibration: asks for an
   icon the plugin has not parsed yet — hence the shortcode text on the first
   visit.  Polling alone only papers over it, because the next reload starts
   cold again.  So the moment a glyph does resolve it is written to
   localStorage, and from then on every first paint is served from there and
   never waits on the plugin at all.  Unknown codes and emoji fall through to
   dc.Markdown exactly as before. */
const SHORTCODE = /^:([A-Za-z0-9_+-]+):$/;
const ICON_KEY = "tkd-icon:";
const ICON_MEM = new Map();

function iconCached(name) {
  if (ICON_MEM.has(name)) return ICON_MEM.get(name);
  let v = null;
  try { v = localStorage.getItem(ICON_KEY + name); } catch (e) { /* ignore */ }
  if (v) ICON_MEM.set(name, v);
  return v;
}

function iconStore(name, svg) {
  ICON_MEM.set(name, svg);
  try { localStorage.setItem(ICON_KEY + name, svg); } catch (e) { /* ignore */ }
}

function iconizeApi() {
  try {
    const P = (app.plugins && app.plugins.plugins) || {};
    const p = P["obsidian-icon-folder"] || P["iconize"];
    if (!p) return null;
    if (p.api) return p.api;
    if (typeof p.getApi === "function") return p.getApi();
  } catch (e) { /* ignore */ }
  return null;
}

function svgOf(ic) {
  if (!ic) return null;
  const s = (typeof ic === "string") ? ic
    : (ic.svgElement || ic.svgContent || ic.svg ||
      (ic.icon && (ic.icon.svgElement || ic.icon.svgContent)));
  return typeof s === "string" && s.indexOf("<svg") >= 0 ? s : null;
}

/* Every accessor Iconize has shipped, in order; the first one that answers
   wins.  No doesIconExists() gate — on a half-loaded pack it says "no" for an
   icon getIconByName() would still hand over, and the result is validated
   here anyway. */
function iconAsk(name) {
  const api = iconizeApi();
  if (!api) return null;
  const tries = [api.getIconByName, api.getSVGFromIconName,
  api.util && api.util.getIconByName];
  for (let i = 0; i < tries.length; i++) {
    if (typeof tries[i] !== "function") continue;
    try {
      const s = svgOf(tries[i].call(api, name));
      if (s) return s;
    } catch (e) { /* try the next one */ }
  }
  return null;
}

function iconizeSvg(name) {
  if (!name) return null;
  const hit = iconCached(name);
  if (hit) return hit;
  const svg = iconAsk(name);
  if (svg) iconStore(name, svg);
  return svg;
}

function useIconSvg(name) {
  const [svg, setSvg] = dc.useState(function () { return iconizeSvg(name); });
  dc.useEffect(() => {
    if (!name || svg) return;
    let live = true, id = 0, n = 0;
    const poll = () => {
      if (!live) return;
      const s = iconizeSvg(name);
      if (s) { setSvg(s); return; }
      if (++n > 60) return;                    // ~25s, then leave the fallback alone
      id = setTimeout(poll, n < 10 ? 150 : 500);
    };
    poll();
    return () => { live = false; clearTimeout(id); };
  }, [name, svg]);
  return svg;
}

function Icon({ code, size }) {
  const px = (size || 14) + "px";
  const box = { display: "inline-flex", alignItems: "center", justifyContent: "center",
                width: px, height: px, flex: "0 0 auto", color: "var(--text-muted)" };
  const m = String(code || "").match(SHORTCODE);
  const svg = useIconSvg(m ? m[1] : null);
  if (svg) return <span class="tkd-i" style={box} dangerouslySetInnerHTML={{ __html: svg }} />;
  if (m) return <span class="tkd-i" style={box}><dc.Markdown content={code} inline={true} /></span>;
  return <span style={{ ...box, fontSize: px }}>{code}</span>;
}

  const stance  = (stanceF ? get(stanceF.key) : 0) + live;
  const frac    = Math.min(1, stance / (STANCE_GOAL * 60));
  const ring    = run ? "var(--color-red)"
                : frac >= 1 ? "var(--color-green)" : "var(--interactive-accent)";

  const timerTile = f => (
    <div key={f.key}
         onClick={() => {
           if (!run) return setRun(true);
           setRun(false);
           if (live) bump(f, live);
           setLive(0);
         }}
         onContextMenu={e => { e.preventDefault(); e.stopPropagation(); if (!run) bump(f, -60); }}
         title={run ? "click to stop and log" : "click to start a hold · right-click −1m"}
         style={{ display: "flex", alignItems: "center", gap: "9px", padding: "3px 6px 3px 3px",
                  cursor: "pointer", userSelect: "none" }}>
      <div style={{ width: "38px", height: "38px", borderRadius: "50%", display: "grid",
                    placeItems: "center", flexShrink: 0,
                    background: `conic-gradient(${ring} ${frac * 360}deg, var(--background-modifier-border) 0)` }}>
        <div style={{ width: "30px", height: "30px", borderRadius: "50%", display: "grid",
                      placeItems: "center", background: "var(--background-secondary)",
                      color: "var(--text-muted)" }}>
          <Icon code={icon(f)} size={13} />
        </div>
      </div>
      <span style={{ display: "flex", flexDirection: "column", gap: "7px" }}>
        <span style={{ ...numStyle, color: run ? "var(--color-red)" : numStyle.color }}>
          {mmss(stance)}
        </span>
        <span style={capStyle}>{f.short}</span>
      </span>
    </div>
  );

  return (
    <div style={{ fontFamily: "var(--font-interface)" }}>
      <style dangerouslySetInnerHTML={{ __html: ICON_CSS }} />
      <div style={{ display: "flex", alignItems: "stretch", flexWrap: "wrap", gap: "12px",
                    padding: "9px 13px", borderRadius: "10px",
                    border: "1px solid var(--background-modifier-border)",
                    background: "var(--background-secondary)" }}>
        {FIELDS.map(f => [
          f.divide ? <Rule key={f.key + "-r"} /> : null,
          f.timer
            ? timerTile(f)
            : <Stat key={f.key} icon={icon(f)} name={f.short}
                    text={show(f, get(f.key))} onBump={d => bump(f, d * step(f))} />,
        ])}
      </div>

      <div style={{ marginTop: "12px" }}>
        <dc.Markdown content={gantt} inline={false} />
      </div>
    </div>
  );
}
